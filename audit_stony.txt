# === 1. INSTALLATION AUTOMATIQUE DES DÉPENDANCES ===
def install_dependencies():
    required_packages = {
        "flask": "flask",
        "fpdf": "fpdf2",
        "matplotlib": "matplotlib",
        "pywebview": "pywebview"
    }
    
    for module_name, package_name in required_packages.items():
        try:
            __import__(module_name)
        except ImportError:
            print(f"[*] Dépendance manquante détectée : {package_name}. Installation en cours...")
            subprocess.check_call([sys.executable, "-m", "pip", "install", package_name])

install_dependencies()

# === 2. CRÉATION AUTOMATIQUE DU RACCOURCI BUREAU (WINDOWS) ===
def create_desktop_shortcut():
    if sys.platform == "win32":
        try:
            desktop = os.path.join(os.path.expanduser("~"), "Desktop")
            shortcut_path = os.path.join(desktop, "STONY-TECHNOLOGIE Audit.lnk")
            
            if not os.path.exists(shortcut_path):
                script_path = os.path.abspath(sys.argv[0])
                python_exe = sys.executable
                
                ps_cmd = f"""
                $WScriptShell = New-Object -ComObject WScript.Shell
                $Shortcut = $WScriptShell.CreateShortcut("{shortcut_path}")
                $Shortcut.TargetPath = "{python_exe}"
                $Shortcut.Arguments = '"{script_path}"'
                $Shortcut.WorkingDirectory = "{os.path.dirname(script_path)}"
                $Shortcut.Description = "STONY-TECHNOLOGIE - Audit Cybersécurité"
                $Shortcut.Save()
                """
                subprocess.run(["powershell", "-Command", ps_cmd], capture_output=True, text=True)
                print("[+] Raccourci Bureau créé avec succès !")
        except Exception as e:
            print(f"[-] Erreur lors de la création du raccourci : {e}")

create_desktop_shortcut()

# === Extraction des journaux Windows ===
def extract_windows_logs(date):
    try:
        cmd = f'powershell Get-WinEvent -FilterHashtable @{{LogName="Security"; StartTime="{date}"; EndTime="{date} 23:59:59"}} | Select TimeCreated, Id, Message'
        result = subprocess.run(cmd, capture_output=True, text=True)
        return result.stdout.splitlines()
    except:
        return ["Extraction Windows non disponible sur cet OS"]

# === Extraction dynamique Firefox ===
def get_firefox_history_path():
    base_path = os.path.expanduser(r"~\\AppData\\Roaming\\Mozilla\\Firefox\\Profiles")
    if os.path.exists(base_path):
        profiles = glob.glob(os.path.join(base_path, "*.default-release")) + glob.glob(os.path.join(base_path, "*.default"))
        if profiles:
            return os.path.join(profiles[0], "places.sqlite")
    return ""

# === Extraction générique navigateurs ===
def extract_browser_history(db_path, date, firefox=False):
    data = []
    if not db_path or not os.path.exists(db_path): return data
    try:
        conn = sqlite3.connect(db_path)
        cursor = conn.cursor()
        if firefox:
            cursor.execute("SELECT url, title, datetime(last_visit_date/1000000,'unixepoch') FROM moz_places")
        else:
            cursor.execute("SELECT url, title, datetime(last_visit_time/1000000-11644473600,'unixepoch') FROM urls")
        for url, title, visit_time in cursor.fetchall():
            try:
                if visit_time:
                    visit_date = datetime.datetime.strptime(visit_time, "%Y-%m-%d %H:%M:%S").date()
                    if visit_date == date:
                        data.append((url, title, visit_time))
            except: continue
        conn.close()
    except Exception as e:
        print(f"Erreur lecture bdd navigateur: {e}")
    return data

# === Génération graphique ===
def generate_graph(counts, filename):
    labels = ["Windows Logs", "Chrome", "Edge", "Firefox"]
    plt.figure(figsize=(6,6))
    plt.pie(counts, labels=labels, autopct='%1.1f%%', startangle=140)
    plt.savefig(filename)
    plt.close()

# === Génération PDF ===
def generate_pdf(date, win_logs, chrome_logs, edge_logs, firefox_logs, filename="rapport_stony.pdf"):
    pdf = StonyPDF()
    pdf.add_page()

    # Page de garde
    pdf.set_font("Arial", 'B', 24)
    pdf.set_text_color(10, 37, 64)
    pdf.cell(0, 20, "Rapport d'Audit Cybersécurité", ln=True, align="C")
    pdf.set_font("Arial", 'I', 14)
    pdf.set_text_color(255, 107, 53)
    pdf.cell(0, 10, f"Date: {date}", ln=True, align="C")
    pdf.ln(20)

    # Sommaire
    pdf.set_font("Arial", 'B', 16)
    pdf.set_text_color(10, 37, 64)
    pdf.cell(0, 10, "Sommaire", ln=True)
    pdf.set_font("Arial", '', 12)
    pdf.set_text_color(0, 0, 0)
    pdf.multi_cell(0, 8, "- Journaux Windows\n- Historique Chrome\n- Historique Edge\n- Historique Firefox\n- Graphiques")

    # Sections
    def add_section(title, color, logs):
        pdf.add_page()
        pdf.set_font("Arial", 'B', 14)
        pdf.set_fill_color(*color)
        pdf.set_text_color(255, 255, 255)
        pdf.cell(0, 10, title, ln=True, fill=True)
        pdf.set_font("Arial", '', 10)
        pdf.set_text_color(0, 0, 0)
        for log in logs[:50]:
            if isinstance(log, tuple):
                pdf.multi_cell(0, 5, f"{log[2]} - {log[1]} ({log[0]})")
            else:
                pdf.multi_cell(0, 5, str(log))

    add_section("Journaux Windows", (10,37,64), win_logs)
    add_section("Historique Chrome", (0,200,150), chrome_logs)
    add_section("Historique Edge", (255,107,53), edge_logs)
    add_section("Historique Firefox", (128,0,128), firefox_logs)

    # Graphiques
    pdf.add_page()
    counts = [len(win_logs), len(chrome_logs), len(edge_logs), len(firefox_logs)]
    graph_file = "graph.png"
    generate_graph(counts, graph_file)
    if os.path.exists(graph_file):
        pdf.image(graph_file, x=30, y=50, w=150)

    pdf.output(filename)
    return filename

            return "Format de date invalide (YYYY-MM-DD)"

        win_logs = extract_windows_logs(date)

        firefox_logs = extract_browser_history(get_firefox_history_path(), date, firefox=True)
